cmake_minimum_required(VERSION 3.16)
project(taskbox CXX)
set(CMAKE_CXX_STANDARD 17)
file(GLOB TASKBOX_SOURCES src/*.cpp)
add_library(taskbox_lib ${TASKBOX_SOURCES})
target_include_directories(taskbox_lib PUBLIC include)
add_executable(taskbox app/main.cpp)
target_link_libraries(taskbox taskbox_lib)
enable_testing()
file(GLOB TASKBOX_TESTS tests/*.cpp)
add_executable(run_tests ${TASKBOX_TESTS})
target_link_libraries(run_tests taskbox_lib)
add_test(NAME taskbox_tests COMMAND run_tests)
